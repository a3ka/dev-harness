# shellcheck shell=bash
prepare_model_trace() {
  local runtime="${HARNESS_SESSION_HOME:-$HOME}/runtime" raw seed roles role rest source
  mkdir -p "$runtime" || die "не создать runtime трассы: $runtime"
  export HARNESS_MODEL_TRACE="$runtime/model-trace.tsv"
  export HARNESS_MODEL_TRACE_SEED="$runtime/model-trace.seed.tsv"
  raw="$runtime/model-trace.$$.raw"
  seed="$HARNESS_MODEL_TRACE_SEED"
  roles="$raw.roles"
  if bash "$HERE/scripts/models_actual.sh" --tsv >"$raw" 2>"$raw.error"; then
    awk -F'\t' 'NF == 3 && $1 != "" && $1 != "orchestrator" && !seen[$1]++ {print}' \
      "$raw" >"$HARNESS_MODEL_TRACE.new"
  else
    printf 'TRACE_ARCHIVE_NOT_READY: '; cat "$raw.error" >&2
    : >"$HARNESS_MODEL_TRACE.new"
  fi
  rm -f "$raw" "$raw.error"
  if [ -n "${PROJECT:-}" ]; then
    # bootstrapTrace names an authority, never a caller-supplied file path.
    source="$(jq -er 'if has("bootstrapTrace") then .bootstrapTrace == "station" else true end' "$PROJECT/harness.project.json" 2>/dev/null)" || source=false
    if [ "$source" != true ]; then
      rm -f "$HARNESS_MODEL_TRACE.new"
      die 'TRACE_SEED_SOURCE_INVALID: bootstrapTrace должен быть station'
    fi
    if [ ! -s "$HARNESS_MODEL_TRACE.new" ] && [ -s "$HARNESS_MODEL_TRACE" ] && [ ! -e "$seed" ]; then
      cp -- "$HARNESS_MODEL_TRACE" "$HARNESS_MODEL_TRACE.new"
    elif [ ! -s "$HARNESS_MODEL_TRACE.new" ] && [ ! -s "$seed" ]; then
      # Read role NAMES only; model bytes still originate exclusively in JSONL.
      jq -er '.roles | keys[]' "$HERE/config/agent_models.json" >"$roles" \
        || die 'TRACE_SEED_REGISTRY_INVALID: нет реестра ролей станции'
      if HARNESS_SESSION_HOME="$ZONE" bash "$HERE/scripts/models_actual.sh" --tsv >"$raw" 2>"$raw.error"; then
        awk -F'\t' 'NF == 3 && $1 != "" && $1 != "orchestrator" && !seen[$1]++ {print}' \
          "$raw" >"$raw.rows"
        : >"$seed.new"
        while IFS=$'\t' read -r role rest; do
          if grep -Fx -- "$role" "$roles" >/dev/null; then
            printf '%s\t%s\n' "$role" "$rest" >>"$seed.new"
          else
            printf 'TRACE_SEED_ROLE_UNKNOWN: %s\n' "$role" >&2
          fi
        done <"$raw.rows"
      else
        printf 'TRACE_SEED_NOT_READY: '; cat "$raw.error" >&2
        : >"$seed.new"
      fi
      chmod 600 "$seed.new"
      mv -f "$seed.new" "$seed"
      rm -f "$raw" "$raw.error" "$raw.rows" "$roles"
    elif [ -f "$seed" ]; then
      # FILENAME distinguishes an empty first input; NR==FNR does not.
      awk -F'\t' 'FILENAME == ARGV[1] { fact[$1] = 1; next } !($1 in fact)' \
        "$HARNESS_MODEL_TRACE.new" "$seed" >"$seed.new"
      chmod 600 "$seed.new"
      mv -f "$seed.new" "$seed"
    fi
    if [ -s "$seed" ]; then
      cat "$seed" >>"$HARNESS_MODEL_TRACE.new"
    fi
  fi
  chmod 600 "$HARNESS_MODEL_TRACE.new"
  mv -f "$HARNESS_MODEL_TRACE.new" "$HARNESS_MODEL_TRACE"
}
